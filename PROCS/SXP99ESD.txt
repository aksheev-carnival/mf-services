SXP99ESD-INV PROC   

       SXP99-WRITE-SX-INV-FILE. 

           IF ESDR-EXC-TYPE NOT = 'EX'  
              GO TO SXP99-WRITE-SX-INV-FILE-EXIT
           END-IF.  

           IF SXP99-SX-INV-FILE-OPENED NOT = 'Y'
              OPEN OUTPUT SHOREX-INV-FILE   
              IF SXP99-DONT-WRITE-HEADER = 'Y'  
                 CONTINUE   
              ELSE  
                 WRITE SHOREX-INV-REC   FROM SXP99-INV-HEADER   
              END-IF
              MOVE 'Y'                  TO   SXP99-SX-INV-FILE-OPENED   
           END-IF.  

           MOVE SPACES                  TO SXP99-INV-ESDR-CALC-KEY. 

           MOVE ZEROES                  TO SXP99-ESDR-PRESALES-ALLOC
                                           SXP99-ESDR-GROUPS-ALLOC  
                                           SXP99-ESDR-PRESALES-SOLD.

           STRING ESDR-CALC-KEY DELIMITED BY SIZE   
                  ESDR-EXC-TIME DELIMITED BY SIZE   
                  INTO SXP99-INV-ESDR-CALC-KEY. 

           PERFORM VARYING SXP99-ETX FROM 1 BY 1 UNTIL SXP99-ETX > 5
              COMPUTE SXP99-ESDR-PRESALES-SOLD       =  
                      ESDR-PRESALES-SOLD (SXP99-ETX) +  
                      SXP99-ESDR-PRESALES-SOLD  
           END-PERFORM. 

           MOVE ESDR-PRESALES-ALLOC     TO SXP99-ESDR-PRESALES-ALLOC.   
           MOVE ESDR-GROUPS-ALLOC       TO SXP99-ESDR-GROUPS-ALLOC. 

           WRITE SHOREX-INV-REC         FROM SXP99-INV-REC. 

       SXP99-WRITE-SX-INV-FILE-EXIT.
            EXIT.   

 END PROC
