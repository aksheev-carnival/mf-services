              MOVE ZEROES                 TO
                             TDP01-STD-MAX-UPGRADE(TDP01-CSUB)  
              MOVE SPACES                 TO
                                   TDP01-STD-MAX-UPGRADE-X(TDP01-CSUB)  
           ELSE 
              MOVE 'Y'                    TO
                                   TDP01-STD-MAX-UPGRADE-X(TDP01-CSUB)  
           END-IF.  

           IF TDP01-SPOILAGE-CODE-DBL-NI(TDP01-CSUB)   = TDP01-NULL-IND 
              MOVE SPACES                 TO
                                   TDP01-SPOILAGE-CODE-DBL(TDP01-CSUB)  
           END-IF.  

           IF TDP01-SPOILAGE-CODE-TRPL-NI(TDP01-CSUB)  = TDP01-NULL-IND 
              MOVE SPACES                 TO
                                   TDP01-SPOILAGE-CODE-TRPL(TDP01-CSUB)

           IF TDP01-SPOILAGE-CODE-QUAD-NI (TDP01-CSUB) = TDP01-NULL-IND

                                  TDP01-OVERRIDE-JUMP-AMT-X (TDP01-CSUB)
           ELSE

           IF TDP01-EFF-AGG-MAX-UPGRADE-NI (TDP01-CSUB) = TDP01-NULL-IND

           END-IF.

       TDP01-CLOSE-RET-SCAT.

           IF SQLCODE = 100 
              MOVE 'Y'                    TO TDP01-END-OF-SCAT-SW   
           END-IF.  

           EXEC SQL 
                CLOSE SAILING_CATEGORY  
           END-EXEC.

           IF SQLCODE < 0   
              GO TO 9150-RDMS-ERROR 
           END-IF.  

           SUBTRACT   1   FROM TDP01-CSUB GIVING TDP01-SAIL-CAT-CNT.

           GO TO TDP01-RET-ALL-SCAT-EXIT.   

       TDP01-SCAT-SQLERROR. 

           IF SQLCODE = 100 
              MOVE 'Y'                    TO TDP01-SCAT-NOTFND-SW   
           ELSE 
              MOVE SQLCODE                TO TDP01-PROC-SQLCODE 
              MOVE 'Y'                    TO TDP01-SCAT-SQLERR-SW   
           END-IF.  

       TDP01-RET-SGL-SCAT-EXIT. 
           EXIT.

       TDP01-RET-ALL-SCAT-EXIT. 
           EXIT.
    
 END PROC
