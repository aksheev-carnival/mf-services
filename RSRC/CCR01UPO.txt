                                           WS-OBC-DEL-AMT   
                 GO TO 0900-PROCESS-BOOKING-DETAILS 
              END-IF
           END-IF.  

           IF WS-TOT-OBC-AMT NOT > ZEROES   
              GO TO 0900-PROCESS-BOOKING-DETAILS
           END-IF.  

           MOVE 0   TO SUB2.
           MOVE ZERO TO WS-BOOKING-OBC-AMT. 

       1000-PROCESS-BOOKING-OBC.

           ADD 1    TO SUB2.

           IF SUB2 > UPOOB1-CNT (SUB1)  
              IF WS-BOOKING-OBC-AMT > 0 
                 GO TO 1000-CHK-OBC-ELIGIBILITY 
              ELSE  
                 GO TO 0900-PROCESS-BOOKING-DETAILS 
              END-IF
           END-IF.  

           IF (UPOOB1-OBC-ACTION-CODE (SUB1 SUB2) = 'U' OR 'A') OR  
              (UPOOB1-OBC-ACTION-CODE (SUB1 SUB2) <= SPACES)
              ADD UPOOB1-OBC-AMT (SUB1 SUB2) TO WS-BOOKING-OBC-AMT  
           END-IF.  


       1000-DELETE-OBC-REC. 

           IF UPOOB1-OBC-ACTION-CODE (SUB1 SUB2) = 'D'  
              PERFORM 1200-GET-OBC-AMT  
                 THRU 1200-GET-OBC-AMT-EXIT 

              EXEC SQL  
                   DELETE ORDER_OBC_DETAIL  
                    WHERE ORDER_NUM      = :UPORD1-ORDER-NUM AND
                          BOOK_NUM       = :UPOBK1-BOOK-NUM (SUB1) AND  
                       PRESALE_ITEM_TYPE = :UPOOB1-ITEM-TYPE (SUB1 SUB2)
              END-EXEC  

              COMPUTE WS-TOT-OBC-AMT  = WS-TOT-OBC-AMT -
                                           WS-OLD-OBC-AMT   

           END-IF.  

           IF UPOOB1-OBC-ACTION-CODE (SUB1 SUB2) = 'U'  
              PERFORM 1200-GET-OBC-AMT  
                 THRU 1200-GET-OBC-AMT-EXIT 

              COMPUTE  WS-TOTAL-OBC = UPOOB1-OBC-AMT (SUB1 SUB2) -  
                                  WS-OLD-OBC-AMT

              IF WS-TOTAL-OBC > 0   
                 GO TO 1000-PROCESS-BOOKING-OBC 
              END-IF

              COMPUTE WS-TOT-OBC-AMT = WS-TOT-OBC-AMT + WS-TOTAL-OBC

              EXEC SQL UPDATE   
                   ORDER_OBC_DETAIL 
                   SET (OBC_AMT = NULLIF(:UPOOB1-OBC-AMT (SUB1 SUB2),0))
                   WHERE ORDER_NUM      = :UPORD1-ORDER-NUM AND 
                         BOOK_NUM       = :UPOBK1-BOOK-NUM (SUB1) AND   
                      PRESALE_ITEM_TYPE = :UPOOB1-ITEM-TYPE (SUB1 SUB2) 
              END-EXEC  
           END-IF.  


           GO TO 1000-PROCESS-BOOKING-OBC.  

       1000-CHK-OBC-ELIGIBILITY.

           IF (RSP06-IBR-OBC-SAILING-AMT NOT > 0)   
              ADD 1                TO CCMV-ADV-SUB  
              MOVE 0015            TO MSGHDR-ADVISE-CODE
                                      (CCMV-ADV-SUB)
              MOVE 'CCR01UPO'      TO MSGHDR-ADVISE-SOURCE  
                                      (CCMV-ADV-SUB)
              PERFORM CCP75-OUTPUT-CCERRMSG 
                 THRU CCP75-OUTPUT-CCERRMSG-EXIT
              GO TO 9000-END-OF-JOB 
           END-IF.  

           MOVE 0                  TO WS-O-OBC-ENTRY-CNT
                                      WS-ORDER-ENTRY-CNT
                                      WS-ENTRY-CNT. 
           PERFORM 3200-GET-OFFERED-OBC THRU 3200-EXIT  
             VARYING RSP06-PAX-SUB FROM 1 BY 1  
               UNTIL RSP06-PAX-SUB > RSP06-PAX-CNT. 
           IF (WS-O-OBC-ENTRY-CNT NOT > 0)  
              ADD 1                TO CCMV-ADV-SUB  
              MOVE 0015            TO MSGHDR-ADVISE-CODE
                                      (CCMV-ADV-SUB)
              MOVE 'CCR01UPO'      TO MSGHDR-ADVISE-SOURCE  
                                      (CCMV-ADV-SUB)
              PERFORM CCP75-OUTPUT-CCERRMSG 
                 THRU CCP75-OUTPUT-CCERRMSG-EXIT
              GO TO 9000-END-OF-JOB 
           END-IF.  

           PERFORM 3700-REARRANGE-OUTPUT
              THRU 3700-EXIT.   

           PERFORM 3800-FIND-ORDERS 
              THRU 3800-EXIT.   

           PERFORM 3900-APPLY-OBC   
              THRU 3900-EXIT.   

           MOVE 0   TO SUB2.

       1000-ADD-BOOKING-OBC.

           ADD 1    TO SUB2.

           IF SUB2 > UPOOB1-CNT (SUB1)  
              GO TO 0900-PROCESS-BOOKING-DETAILS
           END-IF.  

           MOVE 'N'         TO WS-OBC-MATCH.
           EXEC SQL 
                SELECT TRANSLATED_NAME  
                  FROM ITEM_OBC_MAP_TBL 
                 WHERE ITEM_TYPE = :UPOOB1-ITEM-TYPE (SUB1 SUB2)
                  INTO WS-TRANS-NAME
           END-EXEC.

       1000-UPD-OBC-REC.

           IF UPOOB1-OBC-ACTION-CODE (SUB1 SUB2) = 'U'  
              PERFORM 1200-GET-OBC-AMT  
                 THRU 1200-GET-OBC-AMT-EXIT 

              COMPUTE  WS-TOTAL-OBC = UPOOB1-OBC-AMT (SUB1 SUB2) -  
                                  WS-OLD-OBC-AMT

              IF WS-TOTAL-OBC > 0   
                 PERFORM 1100-GET-AVAIL-OBC THRU 1100-EXIT  
              ELSE  
                 GO TO 1000-ADD-BOOKING-OBC 
              END-IF

              COMPUTE WS-TOT-OBC-AMT = WS-TOT-OBC-AMT + WS-TOTAL-OBC

              EXEC SQL UPDATE   
                   ORDER_OBC_DETAIL 
                   SET (OBC_AMT = NULLIF(:UPOOB1-OBC-AMT (SUB1 SUB2),0))
                   WHERE ORDER_NUM      = :UPORD1-ORDER-NUM AND 
                         BOOK_NUM       = :UPOBK1-BOOK-NUM (SUB1) AND   
                      PRESALE_ITEM_TYPE = :UPOOB1-ITEM-TYPE (SUB1 SUB2) 
              END-EXEC  

              GO TO 1000-ADD-BOOKING-OBC
           END-IF.  

       1000-INSERT-OBC-REC. 

           IF UPOOB1-OBC-ACTION-CODE (SUB1 SUB2) NOT = 'A'  
              GO TO 1000-ADD-BOOKING-OBC
           END-IF.  

           MOVE 'N'   TO  WS-OBC-TYP-FND.   

           MOVE UPOOB1-OBC-AMT (SUB1 SUB2)  TO  WS-TOTAL-OBC.   

           IF WS-TOTAL-OBC > 0  
              PERFORM 1100-GET-AVAIL-OBC THRU 1100-EXIT 
           END-IF.  

           COMPUTE WS-TOT-OBC-AMT = WS-TOT-OBC-AMT + WS-TOTAL-OBC.  

           EXEC SQL 
                SELECT SEQ_NUM  
                  INTO :WS-ORD-SEQ  
                  FROM ORDER_OBC_DETAIL 
                 WHERE ORDER_NUM         = :UPORD1-ORDER-NUM AND
                       BOOK_NUM          = :UPOBK1-BOOK-NUM (SUB1) AND  
                       PRESALE_ITEM_TYPE = :UPOOB1-ITEM-TYPE (SUB1 SUB2)
                 FETCH FIRST 1 ROW ONLY 
           END-EXEC.

           IF SQLCODE < 0   
              ADD 1                       TO CCMV-ADV-SUB   
              MOVE 0029                   TO MSGHDR-ADVISE-CODE 
                                             (CCMV-ADV-SUB) 
              MOVE 'CCR01UPO'             TO MSGHDR-ADVISE-SOURCE   
                                             (CCMV-ADV-SUB) 
              PERFORM CCP75-OUTPUT-CCERRMSG 
                 THRU CCP75-OUTPUT-CCERRMSG-EXIT

              GO TO DPP12-RDMS-ERROR
           END-IF.  

           IF SQLCODE = 100 
              CONTINUE  
           ELSE 
              MOVE 'Y'         TO WS-OBC-TYP-FND
           END-IF.  

           IF WS-OBC-TYP-FND = 'N'  
              MOVE 0    TO WS-SEQ-NUM   
                           WS-OBC-REC-CNT   

              EXEC SQL  
                   SELECT MAX(SEQ_NUM)  
                     INTO :WS-SEQ-NUM   
                     FROM ORDER_OBC_DETAIL  
                    WHERE ORDER_NUM  = :UPORD1-ORDER-NUM
              END-EXEC  

              ADD 1         TO WS-SEQ-NUM   
              EXEC SQL  
                   INSERT INTO ORDER_OBC_DETAIL 
                          (ORDER_NUM
                          ,SEQ_NUM  
                          ,BOOK_NUM 
                          ,OBC_AMT  
                          ,PRESALE_ITEM_TYPE)   
                    VALUES (:UPORD1-ORDER-NUM   
                          ,:WS-SEQ-NUM  
                          ,:UPOBK1-BOOK-NUM (SUB1)  
                          ,NULLIF(:UPOOB1-OBC-AMT (SUB1 SUB2),0)
                          ,NULLIF(:UPOOB1-ITEM-TYPE (SUB1 SUB2),' '))   
              END-EXEC  
           ELSE 
              EXEC SQL  
                   UPDATE ORDER_OBC_DETAIL  
                      SET   OBC_AMT = IFNULL(OBC_AMT,0) +   
                                      :UPOOB1-OBC-AMT (SUB1 SUB2)   
                    WHERE ORDER_NUM = :UPORD1-ORDER-NUM AND 
                            SEQ_NUM = :WS-ORD-SEQ   
              END-EXEC  
           END-IF.  

           GO TO  1000-ADD-BOOKING-OBC. 

       1100-GET-AVAIL-OBC.  
            MOVE WS-TOTAL-OBC TO WS-OBC-REQUIRED.   
            PERFORM VARYING SUB FROM 1 BY 1 
             UNTIL SUB > WS-O-OBC-ENTRY-CNT 
                IF WS-O-TRANSLATED-NAME (SUB) = 
                                         WS-TRANS-NAME OR   
                   WS-O-TRANSLATED-NAME (SUB) = 'GENERAL'   
                   IF WS-O-REMAINING-OBC-AMT (SUB)  >= WS-OBC-REQUIRED  
                      SUBTRACT WS-OBC-REQUIRED FROM 
                                WS-O-REMAINING-OBC-AMT (SUB)
                      MOVE ZERO TO WS-OBC-REQUIRED  
                      MOVE WS-O-OBC-ENTRY-CNT  TO SUB   
                   ELSE 
                      SUBTRACT WS-O-REMAINING-OBC-AMT (SUB) FROM
                               WS-OBC-REQUIRED  
                      MOVE ZERO TO WS-O-REMAINING-OBC-AMT (SUB) 
                   END-IF   
                END-IF  
           END-PERFORM. 

           IF WS-OBC-REQUIRED > 0   
              ADD 1                TO CCMV-ADV-SUB  
              MOVE 0016            TO MSGHDR-ADVISE-CODE
                                      (CCMV-ADV-SUB)
              MOVE 'CCR01UPO'      TO MSGHDR-ADVISE-SOURCE  
                                      (CCMV-ADV-SUB)
              PERFORM CCP75-OUTPUT-CCERRMSG 
                 THRU CCP75-OUTPUT-CCERRMSG-EXIT
              GO TO 9000-END-OF-JOB 
           END-IF.  

       1100-EXIT.   
           EXIT.

       1200-GET-OBC-AMT.
           EXEC SQL 
                SELECT IFNULL(OBC_AMT,0)
                  INTO :WS-OLD-OBC-AMT  
                  FROM ORDER_OBC_DETAIL 
                  WHERE    ORDER_NUM  = :UPORD1-ORDER-NUM AND   
                           BOOK_NUM   = :UPOBK1-BOOK-NUM (SUB1) AND 
                    PRESALE_ITEM_TYPE = :UPOOB1-ITEM-TYPE (SUB1 SUB2)   
                    FETCH FIRST 1 ROW ONLY  
           END-EXEC.

           IF SQLCODE < 0   
              ADD 1                       TO CCMV-ADV-SUB   
              MOVE 0030                   TO MSGHDR-ADVISE-CODE 
                                             (CCMV-ADV-SUB) 
              MOVE 'CCR01UPO'             TO MSGHDR-ADVISE-SOURCE   
                                             (CCMV-ADV-SUB) 
              PERFORM CCP75-OUTPUT-CCERRMSG 
                 THRU CCP75-OUTPUT-CCERRMSG-EXIT

              GO TO DPP12-RDMS-ERROR
           END-IF.  

           IF SQLCODE = 100 
              MOVE 0 TO WS-OLD-OBC-AMT  
           END-IF.  

       1200-GET-OBC-AMT-EXIT.   
           EXIT.

       3000-CALL-RSO23. 
           COPY RSP23INT.   

           MOVE WS-BOOK-NUM            TO RSP23-I-INDV-BOOK-NUM.
           MOVE 'A'                    TO RSP23-I-BOOK-TYPE.
           MOVE ZERO                   TO DPP01-SUB-MSG-CTR.

           MOVE 'Y'                    TO RSP23-I-READ-POBC-IND 
                                          RSP23-I-READ-PAX-IND  
                                          RSP23-I-READ-IFR-IND  
                                          RSP23-I-READ-PICF-IND 
                                          RSP23-I-READ-PXFR-IND 
                                          RSP23-I-READ-PTFR-IND 
                                          RSP23-I-READ-PPR-IND  
                                          RSP23-I-READ-TIR-IND  
                                          RSP23-I-READ-CXP-GUEST-IND
                                          RSP23-I-READ-PPIR-IND 
                                          RSP23-I-READ-WAITLIST-IND 
                                          RSP23-I-READ-CIF-IND  
                                          RSP23-I-READ-IBPIR-IND
                                          RSP23-I-READ-I-TOAR-IND   
                                          RSP23-I-READ-PPHOR-IND
                                          RSP23-I-READ-POBC-IND 
                                          RSP23-I-READ-GST-OFR-IND  
                                          RSP23-I-READ-GST-OFR-QUL-IND  
                                          RSP23-I-READ-IBR-IND. 

           CALL 'RSO23RGC' USING DPP01-COMMON-DATA-FIELDS   
                                 DPP01-COMMON-INFO  
                                 RSP23-RGC-INPUT
                                 RSP06-GUEST-CABIN-INFO.

           IF  DPP01-SUB-MSG-CTR > 0 AND
               DPP01-SUB-MSG-STATUS (1) > 1 
               GO TO DPP12-RETURN-SUBROUTINE-ERROR  
           END-IF.  

       3000-CALL-RSO23-EXIT.
           EXIT.

       3200-GET-OFFERED-OBC.

           IF  RSP06-POBC-EXIST-IND (RSP06-PAX-SUB) = 'Y'   
               PERFORM VARYING POBC-SUB FROM 1 BY 1 
                   UNTIL POBC-SUB > RSP06-POBC-ENTRY-CNT
                                       (RSP06-PAX-SUB)  
                   PERFORM 3400-CHECK-VALID-FOR-PRESALES THRU 3400-EXIT 
                   IF WS-VALID-OBC = 'Y'
                      PERFORM 3600-ADD-TO-OFFERED-OBC THRU 3600-EXIT
                   END-IF   
                END-PERFORM 
           END-IF.  

       3200-EXIT.   
           EXIT.

       3400-CHECK-VALID-FOR-PRESALES.   

           MOVE RSP06-POBC-OBC-CODE (RSP06-PAX-SUB POBC-SUB)
                                    TO OBC-CODE.

           FETCH OBC-MASTER-REC RECORD  
                 ON ERROR GO TO 9100-DATABASE-ERROR.

           MOVE 'N'                 TO WS-VALID-OBC 
           MOVE SPACES              TO WS-TNAME.

           IF OBC-DEPT-CODE(1) NOT > SPACES 
              MOVE 'Y'              TO WS-VALID-OBC 
              MOVE 'GENERAL'        TO OBC-DEPT-CODE(1) 
                                       WS-TNAME 
              GO TO 3400-EXIT   
           END-IF.  

           PERFORM VARYING SUB FROM 1 BY 1 UNTIL SUB > MAP-SUB  
              IF (OBC-DEPT-CODE(1) = WS-OBC-TYPE-1 (SUB)) OR
                 (OBC-DEPT-CODE(1) = WS-OBC-TYPE-2 (SUB)) OR
                 (OBC-DEPT-CODE(1) = WS-OBC-TYPE-3 (SUB))   
                 MOVE 'Y'           TO WS-VALID-OBC 
                 MOVE WS-TRANSLATED-NAME (SUB)  
                                    TO WS-TNAME 
                 MOVE MAP-SUB       TO SUB  
              END-IF
           END-PERFORM. 

       3400-EXIT.   
           EXIT.

       3600-ADD-TO-OFFERED-OBC. 

           IF WS-O-OBC-ENTRY-CNT NOT > 0
              MOVE 1                   TO WS-O-OBC-ENTRY-CNT
              MOVE OBC-DEPT-CODE(1) TO WS-O-OBC-DEPT-CODE   
                                          (WS-O-OBC-ENTRY-CNT)  
              MOVE WS-TNAME            TO WS-O-TRANSLATED-NAME  
                                          (WS-O-OBC-ENTRY-CNT)  
              MOVE OBC-REFUND-IND      TO WS-O-OBC-REFUNDABLE-IND   
                                          (WS-O-OBC-ENTRY-CNT)  
              MOVE RSP06-POBC-SAILING-AMT (RSP06-PAX-SUB POBC-SUB)  
                                       TO WS-O-OFFERED-OBC-AMT  
                                          (WS-O-OBC-ENTRY-CNT)  
                                          WS-O-REMAINING-OBC-AMT
                                          (WS-O-OBC-ENTRY-CNT)  
              MOVE ZEROS               TO WS-O-APPLIED-OBC-AMT  
                                          (WS-O-OBC-ENTRY-CNT)  
              GO TO 3600-EXIT   
           ELSE 
              MOVE 'N'                 TO WS-ENTRY-FOUND
              PERFORM VARYING SUB FROM 1 BY 1   
                  UNTIL SUB > WS-O-OBC-ENTRY-CNT
                  IF (OBC-DEPT-CODE(1) = WS-O-OBC-DEPT-CODE (SUB)) AND  
                     (OBC-REFUND-IND = WS-O-OBC-REFUNDABLE-IND (SUB))   
                     MOVE 'Y'          TO WS-ENTRY-FOUND
                     ADD RSP06-POBC-SAILING-AMT (RSP06-PAX-SUB POBC-SUB)
                                       TO WS-O-OFFERED-OBC-AMT  (SUB)   
                                          WS-O-REMAINING-OBC-AMT (SUB)  
                     MOVE WS-O-OBC-ENTRY-CNT
                                       TO SUB   
                  END-IF
               END-PERFORM  

               IF WS-ENTRY-FOUND = 'N'  
                  ADD 1                  TO WS-O-OBC-ENTRY-CNT  
                  MOVE OBC-DEPT-CODE(1)  TO WS-O-OBC-DEPT-CODE  
                                            (WS-O-OBC-ENTRY-CNT)
                  MOVE WS-TNAME          TO WS-O-TRANSLATED-NAME
                                            (WS-O-OBC-ENTRY-CNT)
                  MOVE OBC-REFUND-IND    TO WS-O-OBC-REFUNDABLE-IND 
                                            (WS-O-OBC-ENTRY-CNT)
                  MOVE RSP06-POBC-SAILING-AMT (RSP06-PAX-SUB POBC-SUB)  
                                         TO WS-O-OFFERED-OBC-AMT
                                            (WS-O-OBC-ENTRY-CNT)
                                            WS-O-REMAINING-OBC-AMT  
                                            (WS-O-OBC-ENTRY-CNT)
                  MOVE ZEROS             TO WS-O-APPLIED-OBC-AMT
                                            (WS-O-OBC-ENTRY-CNT)
               ELSE 
                  GO TO 3600-EXIT   
               END-IF   
           END-IF.  

       3600-EXIT.   
           EXIT.

       3700-REARRANGE-OUTPUT.   
           MOVE ZEROS               TO WS-HCNT. 

      *Grouping non-refundable entries first.   

           PERFORM VARYING SUB FROM 1 BY 1  
               UNTIL SUB > WS-O-OBC-ENTRY-CNT   
               IF WS-O-OBC-REFUNDABLE-IND (SUB) = 'N'   
                  ADD 1             TO WS-HCNT  
                  IF WS-HCNT > 1
                     IF WS-O-OBC-DEPT-CODE (SUB) NOT = 'GENERAL' AND
                        WS-HOLD-OBC-DEPT-CODE (WS-HCNT - 1) = 'GENERAL' 
                        MOVE WS-HOLD-ENTRY (WS-HCNT - 1)
                                    TO WS-HOLD-ENTRY (WS-HCNT)  
                        MOVE WS-O-OBC-ENTRY (SUB)   
                                    TO WS-HOLD-ENTRY (WS-HCNT - 1)  
                     ELSE   
                        MOVE WS-O-OBC-ENTRY (SUB)   
                                    TO WS-HOLD-ENTRY (WS-HCNT)  
                     END-IF 
                  ELSE  
                     MOVE WS-O-OBC-ENTRY (SUB)  
                                   TO WS-HOLD-ENTRY (WS-HCNT)   
                  END-IF
               END-IF   
           END-PERFORM. 

      *Grouping refundable entries. 

           PERFORM VARYING SUB FROM 1 BY 1  
               UNTIL SUB > WS-O-OBC-ENTRY-CNT   
               IF WS-O-OBC-REFUNDABLE-IND (SUB) = 'Y'   
                  ADD 1             TO WS-HCNT  
                  IF WS-HCNT > 1
                     IF WS-O-OBC-DEPT-CODE (SUB) NOT = 'GENERAL' AND
                        (WS-HOLD-OBC-DEPT-CODE (WS-HCNT - 1) = 'GENERAL'
                                         AND
                        WS-HOLD-OBC-REFUNDABLE-IND (WS-HCNT - 1) = 'Y') 
                        MOVE WS-HOLD-ENTRY (WS-HCNT - 1)
                                    TO WS-HOLD-ENTRY (WS-HCNT)  
                        MOVE WS-O-OBC-ENTRY (SUB)   
                                    TO WS-HOLD-ENTRY (WS-HCNT - 1)  
                     ELSE   
                        MOVE WS-O-OBC-ENTRY (SUB)   
                                    TO WS-HOLD-ENTRY (WS-HCNT)  
                     END-IF 
                  ELSE  
                     MOVE WS-O-OBC-ENTRY (SUB)  
                                    TO WS-HOLD-ENTRY (WS-HCNT)  
                  END-IF
               END-IF   
           END-PERFORM. 

           IF WS-HCNT NOT = WS-O-OBC-ENTRY-CNT  
              ADD 1                     TO CCMV-ADV-SUB 
              MOVE 0016                 TO MSGHDR-ADVISE-CODE   
                                           (CCMV-ADV-SUB)   
              MOVE 'CCR01UPO'           TO MSGHDR-ADVISE-SOURCE 
                                           (CCMV-ADV-SUB)   
              PERFORM CCP75-OUTPUT-CCERRMSG 
                 THRU CCP75-OUTPUT-CCERRMSG-EXIT
              GO TO 9000-END-OF-JOB 
           END-IF.  

           PERFORM VARYING SUB FROM 1 BY 1  
               UNTIL SUB > WS-HCNT  
               MOVE WS-HOLD-ENTRY (SUB) 
                                    TO WS-O-OBC-ENTRY (SUB) 
           END-PERFORM. 

       3700-EXIT.   
           EXIT.

       3800-FIND-BV-ORDERS. 

            EXEC SQL
                DECLARE ORDER_OBC_CUR CURSOR
                FOR RETENTION   
                SELECT OOD.ORDER_NUM,   
                       IFNULL(OOD.OBC_AMT,0),   
                       IFNULL (OOD.PRESALE_ITEM_TYPE,' '),  
                       CONVERT(OMS.CREATION_DATETIME, 112), 
                       CONVERT(OMS.CREATION_DATETIME, 108)  
                  FROM ORDER_OBC_DETAIL OOD 
                  JOIN ORDER_MASTER OMS 
                       ON OOD.ORDER_NUM = OMS.ORDER_NUM 
                  WHERE (OOD.BOOK_NUM = :RSP23-I-INDV-BOOK-NUM) 
                      AND (OMS.ORDER_STATUS <> 'C') 

            END-EXEC.   

            EXEC SQL
                 OPEN ORDER_OBC_CUR 
            END-EXEC.   

            IF SQLCODE < 0  
               ADD 1                       TO CCMV-ADV-SUB  
              MOVE 0031                   TO MSGHDR-ADVISE-CODE 
                                             (CCMV-ADV-SUB) 
              MOVE 'CCR01UPO'             TO MSGHDR-ADVISE-SOURCE   
                                             (CCMV-ADV-SUB) 
              PERFORM CCP75-OUTPUT-CCERRMSG 
                 THRU CCP75-OUTPUT-CCERRMSG-EXIT

              GO TO DPP12-RDMS-ERROR
            END-IF. 

       3800-PROCESS-BV. 

            EXEC SQL
                 FETCH ORDER_OBC_CUR
                 INTO :WS-ORD-NUM,  
                      :WS-ORD-OBC-AMT,  
                      :WS-ORD-ITEM-TYPE,
                      :WS-ORD-CREATE-DATE,  
                      :WS-TIME  
            END-EXEC.   

            IF SQLCODE < 0 OR SQLCODE = 100 
               GO TO 3800-ORDER-OBC-ERROR   
            END-IF. 
            IF WS-ORD-OBC-AMT > 0   
               ADD 1                       TO WS-ENTRY-CNT

           MOVE WS-GL-ACCT-NUM      TO ARP01-GLCR-NUM.  
           MOVE SSP20-O-SHIP-NUM    TO ARP01-GLCR-SHIP-NUM. 
           MOVE WS-PAYMENT-AMT      TO ARP01-ORIG-TRANSACTION-AMT.  

           COMPUTE ARP01-PAYMENT-AMT ROUNDED =  
                   WS-PAYMENT-AMT / WS-PRC-XCHG-RATE
           COMPUTE ARP01-USD-EQUIVALENT-AMT ROUNDED =   
                   ARP01-PAYMENT-AMT / WS-CLSD-XCHG-RATE
           IF ARP01-PAYMENT-AMT NOT = 0 
              PERFORM ARP10-STORE-CASH-ENTRIES THRU ARP10-EXIT  
           END-IF.  
           IF WS-SURCHARGE-AMT-DIFF NOT = 0 
              EXEC SQL  
                 SELECT  IFNULL (ITEM_LEGCY_SHIP_NUM, ' '), 
                         IFNULL (ITEM_LEGCY_CENTER_NUM, ' '),   
                         IFNULL (ITEM_LEGCY_ACCT_NUM, ' '), 
                         IFNULL (ITEM_LEGCY_SUB_ACCT_NUM,' ')   
                  INTO  :WS-HYB-SRVCH-SHIP-NUM, 
                        :WS-HYB-SRVCH-CENTER-NUM,   
                        :WS-HYB-SRVCH-GL-ACCT-NUM,  
                        :WS-HYB-SRVCH-SUB-ACCT-NUM  
                 FROM    BV_ITEM_TYPE_CGFS_ACCTS
                 WHERE   ITEM_TYPE =  'SRVCH'   
                   AND   COMP_IND = 'N' 
                 FETCH FIRST 1 ROW ONLY 
              END-EXEC  

              IF SQLCODE = 100  
                 CONTINUE   
              END-IF

              MOVE WS-HYB-SR-GL-ACCT-NUM   TO ARP01-GLCR-NUM
              MOVE SSP20-O-SHIP-NUM        TO ARP01-GLCR-SHIP-NUM   
              MOVE WS-SURCHARGE-AMT-DIFF   TO ARP01-ORIG-TRANSACTION-AMT
              COMPUTE ARP01-PAYMENT-AMT ROUNDED =   
                      WS-SURCHARGE-AMT-DIFF / WS-PRC-XCHG-RATE  
              COMPUTE ARP01-USD-EQUIVALENT-AMT ROUNDED =
                      ARP01-PAYMENT-AMT / WS-CLSD-XCHG-RATE 
              PERFORM ARP10-STORE-CASH-ENTRIES THRU ARP10-EXIT  
           END-IF.  

       4200-EXIT.

       4500-UPDATE-HISTORY. 
           MOVE SPACES TO WS-GROUP-BOOK-NUM.

           EXEC SQL 
              SELECT IFNULL(GROUP_BOOK_NUM,' ') 
               INTO  WS-GROUP-BOOK-NUM  
               FROM  CABIN_BOOK 
              WHERE  SHIP_CODE = :WS-SHIP-CODE AND  
                     BOOK_NUM = :WS-BOOK-NUM
           END-EXEC.

           IF SQLCODE < 0   
              ADD 1                       TO CCMV-ADV-SUB   
              MOVE 0033                   TO MSGHDR-ADVISE-CODE 
                                             (CCMV-ADV-SUB) 
              MOVE 'CCR01UPO'             TO MSGHDR-ADVISE-SOURCE   
                                             (CCMV-ADV-SUB) 
              PERFORM CCP75-OUTPUT-CCERRMSG 
                 THRU CCP75-OUTPUT-CCERRMSG-EXIT

              GO TO DPP12-RDMS-ERROR
           END-IF.  

           IF SQLCODE = 100 
              MOVE WS-BOOK-NUM  TO WS-GROUP-BOOK-NUM
           END-IF.  

           MOVE WS-GROUP-BOOK-NUM TO GBR1-BOOK-NUM  

           FETCH GRP1-BOOKING-REC RECORD
               ON ERROR GO TO 4500-CHECK-GBR1-ERROR.

           FETCH LAST GRP1-BOOK-HISTORY-REC 
                    WITHIN GRP1-BOOK-HISTORY-SET SET
               ON ERROR GO TO 4500-CHECK-GBHR-ERROR.

           GO TO 4500-PROCESS-GBHR. 

       4500-CHECK-GBR1-ERROR.   

           IF ERROR-NUM = '0007' OR '0013'  
              GO TO 4500-EXIT   
           END-IF.  

           GO TO 9100-DATABASE-ERROR.   

       4500-CHECK-GBHR-ERROR.   

           IF ERROR-NUM = '0007' OR '0013'  
              GO TO 4500-EXIT   
           END-IF.  

           GO TO 9100-DATABASE-ERROR.   

       4500-PROCESS-GBHR.   

           MOVE 'N'  TO WS-HIST-UPDATE-IND WS-HIST-DONE-IND.

           PERFORM VARYING WS-HSUB FROM GBHR1-ENTRY-CNT BY -1   
               UNTIL WS-HSUB < 1
               IF GBHR1-DATE (WS-HSUB) < WS-SYSTEM-DATE 
                  MOVE 'Y'        TO WS-HIST-DONE-IND   
                  MOVE 1          TO  WS-HSUB   
               END-IF   
               IF GBHR1-ACTIVITY-CODE (WS-HSUB) = 'SSA' 
                  PERFORM VARYING WS-SUB FROM 1 BY 1
                     UNTIL WS-SUB > 40 OR   
                           GBHR1-COMMENT (WS-HSUB) (WS-SUB:1) = ':' 
                     CONTINUE   
                  END-PERFORM   
                  IF WS-SUB < 40
                     ADD 2 TO WS-SUB
                     IF GBHR1-COMMENT (WS-HSUB) (WS-SUB:12)  =  
                         UPORD1-ORDER-NUM   
                        MOVE UPORD1-HYBRIS-ORDER-ID TO WS-HYB-ORD-ID-Z  
                        MOVE WS-HYB-ORD-ID-Z        TO  
                                  GBHR1-COMMENT (WS-HSUB) (WS-SUB: 12)  
                        MOVE 'Y'            TO WS-HIST-UPDATE-IND   
                     END-IF 
                  END-IF
               END-IF   
           END-PERFORM. 

           IF WS-HIST-UPDATE-IND = 'Y'  
              MODIFY GRP1-BOOK-HISTORY-REC RECORD   
           END-IF   

           IF WS-HIST-DONE-IND = 'Y'
              GO TO 4500-EXIT   
           END-IF.  
           FETCH PRIOR GRP1-BOOK-HISTORY-REC
                    WITHIN GRP1-BOOK-HISTORY-SET SET
               ON ERROR GO TO 4500-CHECK-GBHR-ERROR.

           GO TO 4500-PROCESS-GBHR.
