SELECT SA.SHIP_CODE,
       SA.SAIL_ID,
       SA.SAIL_DATE,
       SA.EMBARK_PORT_CODE,
       SA.ITINERARY_CODE
FROM SAILING SA
WHERE SA.SAIL_ID IN (25195,25196,22843,22844,22845,22846,22675,22676,24747,24748,24749,24750,24751,24752,24753,25212,25213,24886,24887,25744,27090,26075,26076,27088,27089)
  AND NOT EXISTS
      (SELECT 1
       FROM ITINERARY_DETAILS ID
       JOIN PORT P
         ON P.PORT_CODE = ID.PORT_CODE
       WHERE ID.EMBARK_PORT_CODE = SA.EMBARK_PORT_CODE
         AND ID.DURATION_DAYS = SA.DURATION_DAYS
         AND ID.ITINERARY_CODE = SA.ITINERARY_CODE
         AND P.COUNTRY_CODE IN ('BB','PR','RU','GD',
         'AU','FR','UY','TR','VU','US','NC','EE','CL',
         'GR','GB','VI','PE','NL','IT','ES','AR','CU',
         'EC','HR','LC','FI','DK','BR','PT','DE'))
ORDER BY SA.SAIL_DATE;
