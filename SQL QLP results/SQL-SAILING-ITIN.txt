       SA.DURATION_DAYS,

LEFT JOIN ITINERARY IT
       ON IT.EMBARK_PORT_CODE = SA.EMBARK_PORT_CODE
      AND IT.DURATION_DAYS = SA.DURATION_DAYS
      AND IT.ITINERARY_CODE = SA.ITINERARY_CODE
