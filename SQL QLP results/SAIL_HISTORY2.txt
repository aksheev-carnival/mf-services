ACTIVITY_CODE, ENTITY_TYPE, ENTITY_CODE, CATEGORY_CODE, 
COMMENT FROM SAIL_HISTORY WHERE SHIP_CODE = 'AQ' 
AND SAIL_ID = 22844 AND METHOD_OF_CHANGE <> 'CSU' 
AND ((ACTIVITY_DATETIME >= '2026-07-08 00:00:00' 
AND ACTIVITY_DATETIME < '2026-07-13 00:00:00') OR 
(ACTIVITY_DATETIME >= '2026-09-21 10:00:00' AND 
ACTIVITY_DATETIME < '2026-09-21 13:00:00')) 
ORDER BY ACTIVITY_DATETIME;
