IRP04CBC PROC   

       IRP04-GET-PASSOFF-DATA.  

           EXEC SQL WHENEVER SQLERROR  CONTINUE END-EXEC.   
           EXEC SQL WHENEVER NOT FOUND CONTINUE END-EXEC.   

           MOVE SPACES                  TO IRP04-MISC-ALPHA-VAR.

           MOVE ZEROES                  TO IRP04-MISC-NUM-VAR.  

           MOVE 'N'                     TO IRP04-CRUO-XFR-SW
                                           IRP04-FLY-XFR-SW 
                                           IRP04-AS-AGREED-IND  
                                           IRP04-PS-TRANSFER-IND
                                           IRP04-PP-GRATS-AVAIL-IND 
                                           IRP04-SKIP-INSTALMENTS-DONE. 

           MOVE 'N'                     TO IRP04-O-AS-AGREED-IND
                                           IRP04-O-SEND-NOTES   
                                           IRP04-O-US-PORT-IND  
                                           IRP04-O-UK-LOCALITY  
                                           IRP04-O-CUBA-SAIL-IND
                                           IRP04-O-AGE-RESTRICTION  
                                           IRP04-O-MSG-OPT-PKG-IND  
                                           IRP04-O-AVL-CAB-VIEW-IND 
                                           IRP04-O-AGENT-DIRECT-IND 
                                           IRP04-O-CATEGORY-MATCH-IND   
                                           IRP04-O-CAB-DISABILITY-IND   
                                           IRP04-O-APPLY-SPOILAGE-IND   
                                           IRP04-O-PP-GRATS-AVAIL-IND   
                                           IRP04-O-EARLY-CABIN-ACC-IND  
                                           IRP04-O-AGENT-INTL-SALES-IND.

           MOVE LOW-VALUES              TO IRP04-PENALTY-INFO.  
           COPY SSP09INT.   

           IF  IRP04-I-INDV-BOOK-NUM NOT > SPACES   
               GO TO IRP04-GET-PASSOFF-DATA-EXIT
           END-IF.  

           IF  IRP04-I-VIEW-TYPE NOT = 'A' AND 'C'  
               GO TO IRP04-GET-PASSOFF-DATA-EXIT
           END-IF.  

       IRP04-CALL-RSO23RGC. 

           COPY RSP06INT.   
           COPY RSP23INT.   
           INITIALIZE RSP06-GUEST-CABIN-INFO.   
           MOVE ZEROES                  TO DPP01-SUB-MSG-CTR.   
           MOVE 'A'                     TO RSP23-I-BOOK-TYPE.   
           MOVE 'Y'                     TO RSP23-I-READ-PAX-IND 
                                           RSP23-I-READ-IBR-IND 
                                           RSP23-I-READ-PICF-IND
                                           RSP23-I-READ-PPHOR-IND   
                                           RSP23-I-READ-PXFR-IND
                                           RSP23-I-READ-POBC-IND
                                           RSP23-I-READ-CIF-IND 
                                           RSP23-I-READ-GST-OFR-QUL-IND.
           MOVE IRP04-I-INDV-BOOK-NUM   TO RSP23-I-INDV-BOOK-NUM.   

           CALL 'RSO23RGC' USING DPP01-COMMON-DATA-FIELDS   
                                 DPP01-COMMON-INFO  
                                 RSP23-RGC-INPUT
                                 RSP06-GUEST-CABIN-INFO.

           IF  DPP01-SUB-MSG-CTR > 0 AND DPP01-SUB-MSG-STATUS (1) > 1   
               DISPLAY 'ERROR FROM RSO23RGC : ' RSP23-I-INDV-BOOK-NUM   
                        UPON PRINTER
               DISPLAY 'DPP01-SUB-MSG-STATUS: ' DPP01-SUB-MSG-STATUS (1)
                      '/DPP01-SUB-MSG-SOURCE: ' DPP01-SUB-MSG-SOURCE (1)
                      '/DPP01-SUB-MSG-INDEX: '  DPP01-SUB-MSG-INDEX  (1)
                        UPON PRINTER
               GO TO DPP12-RETURN-SUBROUTINE-ERROR  
           END-IF.  

       IRP04-FETCH-SAIL.

           MOVE RSP06-IBR-SHIP-CODE     TO TDP01-I-SHIP-CODE.   
           MOVE RSP06-IBR-SAIL-DATE     TO TDP01-I-SAIL-DATE-YYYYMMDD.  
           MOVE RSP06-IBR-DURATION-DAYS TO TDP01-I-DUR-DAYS.

           PERFORM TDP01-RET-SAIL-DETAILS   
              THRU TDP01-RET-SAIL-DETAILS-EXIT. 

           IF  TDP01-SAIL-SQLERR OR 
               TDP01-PROC-ERROR > ZEROES
               GO TO 9150-RDMS-ERROR
           END-IF.  

           IF  TDP01-SAIL-NOTFND OR 
               TDP01-PROC-ERROR > ZEROES
               GO TO 9150-RDMS-ERROR
           END-IF.  

           IF  TDP01-SCAT-SAIL-ID = TDP01-SAIL-ID AND   
               TDP01-SAIL-CAT-CNT > 1   
               CONTINUE 
           ELSE 
               PERFORM TDP01-RET-ALL-SCAT   
                  THRU TDP01-RET-ALL-SCAT-EXIT  
           END-IF.  

           IF  TDP01-SCAT-SQLERR
               GO TO 9150-RDMS-ERROR
           END-IF.  

           MOVE TDP01-SAIL-DATE-YYYYMMDD
                                        TO IRP04-O-SAIL-DATE.   
           MOVE TDP01-SHIP-CODE         TO IRP04-O-SHIP-CODE.   
           MOVE TDP01-DUR-DAYS          TO IRP04-O-SAIL-DURATION-DAYS.  
           MOVE TDP01-SAIL-CAT-CNT      TO IRP04-O-SAIL-CAT-CNT.
           MOVE TDP01-COMPANY-CODE      TO IRP04-COMPANY-CODE.  
           MOVE TDP01-FINANCIAL-COMPANY-CODE
                                        TO IRP04-FIN-COMP-CODE. 
           MOVE TDP01-EMBARK-PORT-CODE  TO PORT-CODE.   
           FETCH PORT-NAME-REC RECORD   
                 ON ERROR GO TO IRP04-PORT-ERROR.   

           IF  PORT-COUNTRY-CODE = 'US' OR 'PR' OR 'BB' 
               MOVE 'Y'                 TO IRP04-O-US-PORT-IND  
           END-IF.  

           MOVE PORT-COUNTRY-CODE       TO IRP04-O-COUNTRY-CODE.

           GO TO IRP04-GET-LOCALITY.

       IRP04-PORT-ERROR.

           IF  ERROR-NUM = '0006' OR '0007' OR '0013'   
               CONTINUE 
           ELSE 
               GO TO 9100-DATABASE-ERROR
           END-IF.  

      * Call RSC86GLD for locality data.
       IRP04-GET-LOCALITY.  

           COPY RSP86INT.   
           MOVE RSP06-IBR-LOCALITY-CODE TO RSP86-I-LOCALITY-CODE.   

           CALL 'RSC86GLD' USING DPP01-COMMON-DATA-FIELDS   
                                 DPP01-COMMON-INFO  
                                 RSP86-INPUT-PARAMS 
                                 RSP86-OUTPUT-PARAMS.   

           IF  DPP01-SUB-MSG-CTR > ZEROES   
               IF  DPP01-SUB-MSG-STATUS (1) > 1 
                   GO TO 9500-RETURN-SUBROUTINE-ERROR   
               END-IF   
           END-IF.  

           IF  RSP86-O-INCL-ATOL-ABTA-INF-IND = 1   
               MOVE 'Y'                 TO IRP04-O-UK-LOCALITY  
           END-IF.  

           MOVE RSP86-O-TIME-ZONE-CODE  TO IRP04-O-TIME-ZONE-CODE.  

       IRP04-PROCESS-CFR-RECORDS.   

           MOVE 'N'                     TO IRP04-CFR-BK-IND 
                                           IRP04-CFR-PAY-IND.   

           MOVE RSP06-IBR-BOOK-NUM      TO CFR-BOOK-NUM.
           FETCH CONFIRMATION-REC RECORD
                 ON ERROR GO TO IRP04-CFR-ERROR.

           IF  CFR-ACTIVITY-CODE = 'CRC'
               MOVE 'Y'                 TO IRP04-CFR-PAY-IND
           ELSE 
               IF  CFR-ACTIVITY-CODE = 'ACB' OR 'NBK' OR 'PNI' OR 'PNA' 
                   MOVE 'Y'             TO IRP04-CFR-BK-IND 
               ELSE 
                   CONTINUE 
               END-IF   
           END-IF.  

           IF  IRP04-CFR-PAY-IND = 'Y' AND IRP04-CFR-BK-IND = 'Y'   
               MOVE 'B'                 TO IRP04-O-CFR-COND 
           ELSE 
               IF  IRP04-CFR-PAY-IND = 'Y' AND IRP04-CFR-BK-IND = 'N'   
                   MOVE 'P'             TO IRP04-O-CFR-COND 
               ELSE 
                   IF  IRP04-CFR-PAY-IND = 'N' AND  
                                               IRP04-CFR-BK-IND = 'Y'   
                       MOVE 'N'         TO IRP04-O-CFR-COND 
                   END-IF   
               END-IF   
           END-IF.  

           GO TO IRP04-FETCH-AGENT-DETAILS. 

      * Display any errors relating to processing confirmation records  
       IRP04-CFR-ERROR. 

           IF  ERROR-NUM = '0006' OR '0007' OR '0013'   
               CONTINUE 
           ELSE 
               GO TO 9100-DATABASE-ERROR
           END-IF.  

       IRP04-FETCH-AGENT-DETAILS.   

      * Fetch Agent related records.

           MOVE SPACES                  TO AAR-CUSTOMER-KEY.
           IF  RSP06-IBR-ALT-CUSTOMER-NUM > SPACES  
               MOVE RSP06-IBR-ALT-CUSTOMER-NUM  
                                        TO AAR-CUSTOMER-NUM 
           ELSE 
               MOVE RSP06-IBR-CUSTOMER-NUM  
                                        TO AAR-CUSTOMER-NUM 
           END-IF.  
           MOVE 'CM'                    TO AAR-CUSTOMER-TYPE.   

           FETCH AGENCY-ASSOCIATION-REC RECORD  
                 ON ERROR GO TO IRP04-AGENT-ERROR.  

           FETCH OWNER RECORD WITHIN AGENCY-ASSOCIATION-SET SET 
                 ON ERROR GO TO IRP04-AGENT-ERROR.  

           IF  AGENT-DIRECT-IND = 'Y'   
               PERFORM VARYING IRP04-PPG-SUB FROM 1 BY 1
                         UNTIL IRP04-PPG-SUB > RSP06-PAX-CNT
                  IF  RSP06-PAX-PP-GRATS-IND (IRP04-PPG-SUB) = 'Y'  
                      MOVE 'Y'          TO IRP04-PP-GRATS-AVAIL-IND 
                      MOVE RSP06-PAX-CNT
                                        TO IRP04-PPG-SUB
                  END-IF
               END-PERFORM  
               MOVE IRP04-PP-GRATS-AVAIL-IND
                                        TO IRP04-O-PP-GRATS-AVAIL-IND   
               MOVE 'KHSCAS'            TO IRP04-O-AGENCY-ID
               MOVE RSP06-IBR-AGENT-CONTACT 
                                        TO IRP04-O-AGENCY-GUEST-NAME
           ELSE 
               MOVE AGENT-NAME          TO IRP04-O-AGENCY-GUEST-NAME
               MOVE AGENT-ABN-NUMBER    TO IRP04-O-AGENT-ABN-NUM
               MOVE AGENT-CUSTOMER-NUM  TO IRP04-O-AGENCY-ID
               MOVE AGENT-INTL-SALES-IND
                                        TO IRP04-O-AGENT-INTL-SALES-IND 
               MOVE AGENT-COUNTRY-CODE  TO IRP04-O-AGENT-COUNTRY-CODE   
               MOVE AGENT-ABTA-NUM      TO IRP04-O-ABTA-NUM 
               MOVE AGENT-TTA-NUM       TO IRP04-O-TTA-NUM  
               PERFORM VARYING IRP04-ATOL-SUB FROM 1 BY 1   
                         UNTIL IRP04-ATOL-SUB > AGENT-BIN-TABLE-COUNT   
                  IF  AGENT-BUS-ID-NUM-TYP (IRP04-ATOL-SUB) = 'ATOL'
                      MOVE AGENT-BUS-ID-NUM-VAL (IRP04-ATOL-SUB)
                                        TO IRP04-O-ATOL-NUM 
                  END-IF
               END-PERFORM  
           END-IF.  

           MOVE AGENT-DIRECT-IND        TO IRP04-O-AGENT-DIRECT-IND.
           MOVE AGENT-PHONE-NUM         TO IRP04-O-PHONE-NUM.   
           MOVE AGENT-ADDRESS1          TO IRP04-O-ADDRESS-1.   
           MOVE AGENT-ADDRESS2          TO IRP04-O-ADDRESS-2.   
           MOVE AGENT-CITY              TO IRP04-O-CITY.
           MOVE AGENT-STATE             TO IRP04-O-STATE.   
           MOVE AGENT-ZIP               TO IRP04-O-ZIP-CODE.
           MOVE RSP06-IBR-AGENT-CONTACT TO IRP04-O-AGENT-CONTACT.   

           GO TO IRP04-GET-PACKAGES.

       IRP04-AGENT-ERROR.   

           IF  ERROR-NUM = '0006' OR '0007' OR '0013'   
               CONTINUE 
           ELSE 
               GO TO 9100-DATABASE-ERROR
           END-IF.  

       IRP04-GET-PACKAGES.  

      * Logic for booking with packages.
           IF  RSP06-IBR-PRE-PKG-NUM > SPACES   
               MOVE RSP06-IBR-PRE-PKG-NUM   
                                        TO TPR-PKG-NUM  
               FETCH TOUR-PACKAGE-REC RECORD
                     ON ERROR GO TO IRP04-TPR-ERROR 
               MOVE TPR-NUM-CAR         TO IRP04-O-NUM-CAR  
               MOVE TPR-PKG-DAYS        TO IRP04-O-PKG-DAYS 
               MOVE TPR-NUM-XFR         TO IRP04-O-NUM-XFR  
               PERFORM DPP20-UNLOAD-TPR-DATES   
                  THRU DPP20-EXIT   
               MOVE TPR-PKG-CODE        TO DPP30-PKG-CODE   
               MOVE ZEROES              TO DPP30-PKG-PHASE-NUM  
               PERFORM DPP30-FETCH-PDR-REC  
                  THRU DPP30-EXIT   
               MOVE PDR1-DESCRIPTION    TO IRP04-O-PACKAGE-DESC 
               MOVE 'Y'                 TO IRP04-O-MSG-OPT-PKG-IND  
           END-IF.  

           IF  RSP06-IBR-PST-PKG-NUM > SPACES   
               MOVE RSP06-IBR-PST-PKG-NUM   
                                        TO TPR-PKG-NUM  
               FETCH TOUR-PACKAGE-REC RECORD
                     ON ERROR GO TO IRP04-TPR-ERROR 
               MOVE TPR-NUM-CAR         TO IRP04-O-NUM-CAR  
               MOVE TPR-PKG-DAYS        TO IRP04-O-PKG-DAYS 
               MOVE TPR-NUM-XFR         TO IRP04-O-NUM-XFR  
               PERFORM DPP20-UNLOAD-TPR-DATES   
                  THRU DPP20-EXIT   
               MOVE TPR-PKG-CODE        TO DPP30-PKG-CODE   
               MOVE ZEROES              TO DPP30-PKG-PHASE-NUM  
               PERFORM DPP30-FETCH-PDR-REC  
                  THRU DPP30-EXIT   
               MOVE PDR1-DESCRIPTION    TO IRP04-O-PACKAGE-DESC 
           END-IF.  

           GO TO IRP04-BUILD-FIELDS.

       IRP04-TPR-ERROR. 

           IF  ERROR-NUM = '0006' OR '0007' OR '0013'   
               CONTINUE 
           ELSE 
               GO TO 9100-DATABASE-ERROR
           END-IF.  

       IRP04-BUILD-FIELDS.  

           MOVE 'E'                     TO IRP04-O-EMAIL-FAX-TYPE.  
           MOVE TDP01-PP-GRATS-APPLY-CODE   
                                        TO IRP04-O-PPGRATS-APPLY-CODE.  
           MOVE TDP01-CUBA-SAIL-IND     TO IRP04-O-CUBA-SAIL-IND.   
           MOVE TDP01-EARLY-CABIN-ACC-IND   
                                        TO IRP04-O-EARLY-CABIN-ACC-IND. 
           MOVE TDP01-ITINERARY-CODE    TO IRP04-O-ITINERARY-CODE.  
           MOVE TDP01-EMBARK-PORT-CODE  TO IRP04-O-EMPARK-PORT-CODE.

           MOVE RSP06-IBR-LOCALITY-CODE TO IRP04-O-LOCALITY-CODE.   
           MOVE RSP06-PAX-CNT           TO IRP04-O-TOTAL-PAX-COUNT. 
           MOVE RSP06-IBR-CURRENCY-CODE TO IRP04-O-CURRENCY-CODE.   
           MOVE RSP06-IBR-FINANCIAL-COMP-CODE   
                                        TO IRP04-O-FINANCIAL-COMP-CODE. 
           COMPUTE IRP04-O-DEPOSIT-AMT = RSP06-IBR-DEPOSIT-AMT +
                                         RSP06-IBR-INS-PREMIUM-AMT. 
           MOVE RSP06-IBR-NET-FARE-AMT  TO IRP04-O-NET-FARE-AMT.
           MOVE RSP06-IBR-AUTOPAY-REQUEST-CODE  
                                        TO IRP04-O-AUTOPAY-REQUEST-CODE.
           MOVE RSP06-IBR-FINAL-DATE    TO IRP04-O-FINAL-DATE.  
           MOVE RSP06-IBR-CABIN-NUM     TO IRP04-CABIN-NUM. 
           MOVE RSP06-IBR-CABIN-NUM(1:1)
                                        TO IRP04-O-CABIN-TYPE.  
           MOVE RSP06-IBR-NUM-ROLLAWAYS TO IRP04-O-NUM-ROLLAWAYS.   
           MOVE RSP06-IBR-NUM-CRIBS     TO IRP04-O-RAB-CRIB-NUM.
           MOVE RSP06-IBR-BKG-COMMENT1  TO IRP04-BKG-COMMENT1.  
           MOVE RSP06-IBR-BKG-COMMENT2  TO IRP04-BKG-COMMENT2.  
           MOVE RSP06-IBR-PAY-COMMENT   TO IRP04-PAY-COMMENT.   
           PERFORM VARYING IRP04-COMM-SUB FROM 1 BY 1   
                     UNTIL IRP04-COMM-SUB > 45  
              IF  IRP04-BKG-COMMENT1(IRP04-COMM-SUB:1) = '|'
                  MOVE ' '              TO IRP04-BKG-COMMENT1   
                                                      (IRP04-COMM-SUB:1)
              END-IF
              IF  IRP04-BKG-COMMENT2(IRP04-COMM-SUB:1) = '|'
                  MOVE ' '              TO IRP04-BKG-COMMENT2   
                                                      (IRP04-COMM-SUB:1)
              END-IF
              IF  IRP04-PAY-COMMENT(IRP04-COMM-SUB:1) = '|' 
                  MOVE ' '              TO IRP04-PAY-COMMENT
                                                      (IRP04-COMM-SUB:1)
              END-IF
           END-PERFORM. 
           MOVE IRP04-BKG-COMMENT1      TO IRP04-O-BKG-COMMENT1.
           MOVE IRP04-BKG-COMMENT2      TO IRP04-O-BKG-COMMENT2.
           MOVE IRP04-PAY-COMMENT       TO IRP04-O-PAY-COMMENT. 
           MOVE RSP06-IBR-PST-PKG-NUM   TO IRP04-O-PST-PKG-NUM. 
           MOVE RSP06-IBR-PRE-PKG-NUM   TO IRP04-O-PRE-PKG-NUM. 
           MOVE RSP06-IBR-GTEE-CATEGORY-CODE
                                        TO IRP04-O-GTEE-CATEGORY-CODE.  
           MOVE RSP06-IBR-PENALTY-ITEM-CODE 
                                        TO IRP04-O-PENALTY-ITEM-CODE.   
           MOVE RSP06-IBR-TRVL-WITH-NUM TO IRP04-O-TRVL-WITH-NUM.   

           MOVE SPACES                  TO IRP04-LOOP-SWITCH
                                           IRP04-CATEGORY-CODE  
                                           IRP04-META-CATEGORY-CODE 
                                           IRP04-META-CATEGORY-NAME.

           IF  RSP06-IBR-GTEE-CATEGORY-CODE > SPACES AND
               IRP04-O-CABIN-TYPE NOT = 'W' 
               MOVE RSP06-IBR-GTEE-CATEGORY-CODE
                                        TO IRP04-CATEGORY-CODE  
           ELSE 
               MOVE RSP06-IBR-CATEGORY-CODE 
                                        TO IRP04-CATEGORY-CODE  
           END-IF.  

           IF  NOT AS-AGREED AND RSP06-IBR-GROUP-BOOK-NUM > SPACES  
               MOVE RSP06-IBR-GROUP-BOOK-NUM
                                        TO GBR1-BOOK-NUM
               FETCH GRP1-BOOKING-REC RECORD
               IF  GBR1-AS-AGREED-IND > 0   
                   MOVE 'Y'             TO IRP04-AS-AGREED-IND  
               END-IF   
           END-IF   

           EXEC SQL 
                SELECT AS_AGREED_IND
                  INTO :IRP04-AS-AGREED-IND 
                  FROM RATE_MASTER  
                 WHERE PRP_OWNER_ID = :RSP06-PAX-RATE-OWNER-ID(1)   
                   AND PRP_OWNER_CODE = 'RTE'   
           END-EXEC.

           IF  SQLCODE < 0  
               GO TO 9150-RDMS-ERROR
           END-IF.  

           IF  AS-AGREED
               MOVE 'Y'                 TO IRP04-O-AS-AGREED-IND
           END-IF.  

           PERFORM VARYING IRP04-TDP-SUB FROM 1 BY 1
                    UNTIL IRP04-TDP-SUB > TDP01-SAIL-CAT-CNT
                    OR END-OF-LOOP  
              IF  TDP01-CATEGORY-CODE (IRP04-TDP-SUB) = 
                                           IRP04-CATEGORY-CODE

                                        TO IRP04-O-TOTAL-TRANS-CHARGES. 
           COMPUTE IRP04-O-TOTAL-CRUISE-CHARGES =   
                   IRP04-TOTAL-CRUISE-CHARGES   +   
                   IRP04-TOTAL-TRANS-CHARGES.   

***********BUILD ITINERARY DETAILS**************************************

           EXEC SQL 
                SELECT IFNULL(SUBREGION_CODE, ' '), 
                       NBK_DESCRIPTION  
                  INTO :IRP04-DEST-CODE,
                       :IRP04-DEST-NAME 
                  FROM ITINERARY
                 WHERE EMBARK_PORT_CODE = :TDP01-EMBARK-PORT-CODE   
                  AND  DURATION_DAYS    = :TDP01-DURATION-DAYS  
                  AND  ITINERARY_CODE   = :TDP01-ITINERARY-CODE 
           END-EXEC.

           IF  SQLCODE < 0  
               GO TO 9150-RDMS-ERROR
           ELSE 
               IF  SQLCODE = 100
                   MOVE SPACES          TO IRP04-DEST-CODE  
                                           IRP04-DEST-NAME  
               END-IF   
           END-IF.  

           MOVE IRP04-DEST-CODE         TO IRP04-O-DESTINATION-CODE.
           MOVE IRP04-DEST-NAME         TO IRP04-O-DESTINATION-NAME.

           EXEC SQL 
                DECLARE ITNDET_TBL_CSR CURSOR   
                FOR RETENTION   
                SELECT ARRIVAL_TIME ,   
                       DAY_NUMBER , 
                       DEPART_TIME ,
                       PORT_CODE,   
                       TENDER_CODE, 
                       FOOTNOTE_CODE
                  FROM ITINERARY_DETAILS
                 WHERE EMBARK_PORT_CODE = :TDP01-EMBARK-PORT-CODE   
                  AND  DURATION_DAYS    = :TDP01-DURATION-DAYS  
                  AND  ITINERARY_CODE   = :TDP01-ITINERARY-CODE 
           END-EXEC.

           EXEC SQL 
                OPEN ITNDET_TBL_CSR 
           END-EXEC.

           IF  SQLCODE < 0  
               GO TO 9150-RDMS-ERROR
           END-IF.  

           MOVE 'N'                     TO IRP04-END-OF-CURSOR-IND. 
           PERFORM VARYING IRP04-ITN-SUB FROM 1 BY 1
                     UNTIL IRP04-END-OF-CURSOR-IND = 'Y'
              MOVE SPACES               TO IRP04-PORT-NAME  
                                           IRP04-ITR-DAY-WEEK   
              MOVE ZEROES               TO IRP04-DAY-NUM
              EXEC SQL FETCH NEXT ITNDET_TBL_CSR
                   INTO :IRP04-O-ITR-ARRIVE-TIME(IRP04-ITN-SUB) 
                        :IRP04-ARRIVAL-TIME-NI, 
                        :IRP04-DAY-NUM, 
                        :IRP04-O-ITR-DEPART-TIME(IRP04-ITN-SUB) 
                        :IRP04-DEPART-TIME-NI,  
                        :IRP04-ITN-PORT-CODE :IRP04-PORT-CODE-NI,   
                        :IRP04-O-ITR-TENDER-CODE(IRP04-ITN-SUB),
                        :IRP04-O-ITR-FOOTNOTE-CODE(IRP04-ITN-SUB)   
                        :IRP04-FOOTNOTE-CODE-NI 
              END-EXEC  
              IF  SQLCODE < 0   
                  GO TO 9150-RDMS-ERROR 
              ELSE  
                  IF  SQLCODE = 100 
                      MOVE 'Y'          TO IRP04-END-OF-CURSOR-IND  
                  ELSE  
                      IF  IRP04-ARRIVAL-TIME-NI = TDP01-NULL-IND
                          MOVE SPACES   TO IRP04-O-ITR-ARRIVE-TIME  
                                                         (IRP04-ITN-SUB)
                      END-IF
                      IF  IRP04-DEPART-TIME-NI = TDP01-NULL-IND 
                          MOVE SPACES   TO IRP04-O-ITR-DEPART-TIME  
                                                         (IRP04-ITN-SUB)
                      END-IF
                      IF  IRP04-PORT-CODE-NI = TDP01-NULL-IND   
                          MOVE SPACES   TO IRP04-O-ITR-PORT-OF-CALL 
                                                         (IRP04-ITN-SUB)
                      END-IF
                      PERFORM IRP04-ITN-PORT-NAME   
                         THRU IRP04-ITN-PORT-NAME-EXIT  
                      MOVE IRP04-PORT-NAME  
                                        TO IRP04-O-ITR-PORT-OF-CALL 
                                                         (IRP04-ITN-SUB)
                      MOVE TDP01-SAIL-DATE-YYYYMMDD 
                                        TO DPP02-YYYYMMDD-DATE  
                      MOVE IRP04-DAY-NUM
                                        TO DPP02-NUMBER-DAYS
                      PERFORM DPP10-ADVANCE-YYYYMMDD-DATE   
                      PERFORM DPP10-YYYYMMDD-WEEK-DAY   
                      PERFORM DPP10-YYYYMMDD-TO-MMDDYYYY
                      MOVE DPP02-DAY-OF-WEEK
                                        TO IRP04-ITR-DAY
                      MOVE DPP02-MMDDYYYY-DATE  
                                        TO IRP04-ITR-DATE   
                      MOVE ', '         TO IRP04-ITR-DAY-COMMA  

                      STRING IRP04-ITR-DAY       DELIMITED BY SPACES
                             IRP04-ITR-DAY-COMMA DELIMITED BY SIZE  
                             IRP04-ITR-DATE      DELIMITED BY SIZE  
                        INTO IRP04-O-ITR-DAY-OF-WEEK (IRP04-ITN-SUB)
                      IF  IRP04-FOOTNOTE-CODE-NI = TDP01-NULL-IND   
                          MOVE SPACES   TO IRP04-O-ITR-FOOTNOTE-CODE
                                                     (IRP04-ITN-SUB)
                      END-IF
                  END-IF
              END-IF
           END-PERFORM. 

           MOVE '3FA7B9C2D85FA7B9C2D8C3' TO IRP04-EOF-FILLER.   

       IRP04-CLOSE-ITN-CURSOR.  

           EXEC SQL 
                CLOSE ITNDET_TBL_CSR
           END-EXEC.

           IF  SQLCODE < 0  
               GO TO 9150-RDMS-ERROR
           END-IF.  

***********BUILD SPLITPAYMENT*******************************************

       IRP04-SPLITPAY-START.

           IF  (RSP06-IBR-AUTOPAY-REQUEST-CODE NOT > SPACES) OR 
               (RSP06-IBR-AUTOPAY-REQUEST-CODE  = 'C' OR 'N')   
               GO TO IRP04-SPLITPAY-FINISH-EXIT 
           END-IF.  

           MOVE 'SPLITPAYMENT'          TO SSP09-I-CODE.
           STRING 'OPT' RSP06-IBR-AUTOPAY-REQUEST-CODE  
                        DELIMITED BY SIZE INTO SSP09-I-TYPE.

           CALL 'SSC09GCD' USING DPP01-COMMON-DATA-FIELDS   
                                 DPP01-COMMON-INFO  
                                 SSP09-INPUT-PARAMS 
                                 SSP09-OUTPUT-PARAMS.   

           IF  DPP01-SUB-MSG-CTR > 0 AND DPP01-SUB-MSG-STATUS (1) > 1   
               IF  (DPP01-SUB-MSG-SOURCE (1) = 'SSC09GCD' AND   
                       DPP01-SUB-MSG-INDEX (1) = 2) OR  
                         (DPP01-SUB-MSG-SOURCE (1) = 'SSC10LCD' AND 
                             DPP01-SUB-MSG-INDEX (1) = 5)   
                    GO TO IRP04-SPLITPAY-FINISH-EXIT
               ELSE 
                    GO TO 9500-RETURN-SUBROUTINE-ERROR  
               END-IF   
           END-IF.  

           IF  RSP06-IBR-AUTOPAY-REQUEST-CODE = 'M' 
               GO TO IRP04-SPLITPAY-BOOKING 
           END-IF.  

           MOVE FUNCTION NUMVAL (SSP09-O-DESCRIPTION)   
                                        TO IRP04-INSTALMENTS-OPTED. 

       IRP04-SPLITPAY-BOOKING.  

           MOVE RSP06-IBR-BOOK-NUM      TO PHR-PRODUCT-NUM. 

           MOVE 'PMTHDR-'               TO PMTHDR-CALCAREA(1:7).

           IF  RSP06-IBR-BOOK-NUM(6:1) NOT NUMERIC  
               MOVE 'S'                 TO PMTHDR-CALCAREA(8:1) 
           ELSE 
               MOVE RSP06-IBR-BOOK-NUM(6:1) 
                                        TO PMTHDR-CALCAREA(8:1) 
           END-IF.  

           FETCH PAYMENT-HEADER-REC RECORD  
                 ON ERROR GO TO IRP04-PAYMENT-ERROR.

           MOVE 0                       TO IRP04-INSTALLMENTS-PROCESSED.

       IRP04-CHECK-INSTALLMENTS.

           FETCH NEXT PAYMENT-REC WITHIN PMT-SET SET
                 ON ERROR GO TO IRP04-INSTALLMENTS-PENDING. 

           IF  PYDR-AUTOPAY-REQUEST-CODE = 'T'  
               MOVE 'Y'                 TO IRP04-SKIP-INSTALMENTS-DONE  
               MOVE ZEROES              TO IRP04-INSTALLMENTS-PROCESSED 
               GO TO IRP04-CHECK-INSTALLMENTS   
           END-IF.  

           IF  (RSP06-IBR-AUTOPAY-REQUEST-CODE  
                                      = PYDR-AUTOPAY-REQUEST-CODE) OR   
               (PYDR-AUTOPAY-REQUEST-CODE = 'Z')
               MOVE 'N'                 TO IRP04-SKIP-INSTALMENTS-DONE  
               IF  IRP04-PLAN-START-DATE NOT > 0
                   MOVE PYDR-PAYMENT-DATE   
                                        TO IRP04-PLAN-START-DATE
                   IF  RSP06-IBR-AUTOPAY-REQUEST-CODE = 'M' 
                       MOVE IBR-FINAL-DATE  
                                        TO IRP04-FINAL-PAYMENT-DATE 
                       COMPUTE IRP04-INSTALMENTS-OPTED  
                             = ((IRP04-FINAL-DATE-YYYY  
                             -  IRP04-PLAN-INSTALLMENT-YYYY)* 12)   
                             +  IRP04-FINAL-DATE-MM 
                             -  IRP04-PLAN-INSTALLMENT-MM   
                   IF (IRP04-FINAL-DATE-DD <= IRP04-PLAN-INSTALLMENT-DD)
                       AND (IRP04-FINAL-DATE-DD NOT = 29 AND 30 AND 31) 
                       SUBTRACT 1 FROM IRP04-INSTALMENTS-OPTED  
                                  GIVING IRP04-INSTALMENTS-OPTED
                   END-IF   
                   END-IF   
               END-IF   
           END-IF.  

           IF  (IRP04-SKIP-INSTALMENTS-DONE = 'N')  AND 
               (PYDR-AUTOPAY-REQUEST-CODE = 'I' OR 'O') 
               CONTINUE 
           ELSE 
               GO TO IRP04-CHECK-INSTALLMENTS   
           END-IF.  

           ADD 1                        TO IRP04-INSTALLMENTS-PROCESSED.

           IF  IRP04-INSTALLMENTS-PROCESSED = IRP04-INSTALMENTS-OPTED   
               MOVE PYDR-PAYMENT-DATE   TO IRP04-LAST-INSTALLMENT-DATE  
               GO TO IRP04-SPLITPAY-FINISH  
           END-IF.  

           GO TO IRP04-CHECK-INSTALLMENTS.  

       IRP04-PAYMENT-ERROR. 

           IF  ERROR-NUM NOT = '0007' AND '0013'
               GO TO 9100-DATABASE-ERROR
           END-IF.  

           GO TO 9200-EDIT-ERROR-MSG.   

       IRP04-INSTALLMENTS-PENDING.  

           IF  ERROR-NUM = '0006' OR '0007' OR '0013'   
               CONTINUE 
           ELSE 
               GO TO 9100-DATABASE-ERROR
           END-IF.  

       IRP04-SPLITPAY-FINISH.   

           MOVE IRP04-INSTALLMENTS-PROCESSED
                                      TO IRP04-O-INSTALLMENTS-PROCESSED.
           MOVE IRP04-INSTALMENTS-OPTED 
                                      TO IRP04-O-INSTALLMENTS-OPTED.
           MOVE IRP04-PLAN-START-DATE TO IRP04-O-PLAN-START-DATE.   
           MOVE IRP04-LAST-INSTALLMENT-DATE 
                                      TO IRP04-O-LAST-INSTALLMENT-DATE. 

       IRP04-SPLITPAY-FINISH-EXIT.  

           GO TO IRP04-GET-PASSOFF-DATA-EXIT.   

       IRP04-ITN-PORT-NAME. 

           EXEC SQL 
                SELECT  NAME
                  INTO  :IRP04-PORT-NAME
                  FROM  PORT
                  WHERE PORT_CODE = :IRP04-ITN-PORT-CODE
                  FETCH FIRST 1 ROW ONLY
           END-EXEC.

           IF  SQLCODE < 0  
               GO TO 9150-RDMS-ERROR
           ELSE 
               IF  SQLCODE = 100
                   GO TO IRP04-ITN-PORT-NAME-EXIT   
               END-IF   
           END-IF.  

       IRP04-ITN-PORT-NAME-EXIT.
           EXIT.

       IRP04-PROCESS-PS-TRANSFERS.  

           MOVE 'Y'                     TO IRP04-PS-TRANSFER-IND.   
           IF  RSP06-PAX-BOOK-TYPE (IRP04-XFER-SUB) = 'I'   
               MOVE RSP06-PAX-BOOK-NUM (IRP04-XFER-SUB) 
                                        TO PTFR-BOOK-NUM
           ELSE 
               MOVE RSP06-PAX-CABIN-BOOK-NUM (IRP04-XFER-SUB)   
                                        TO PTFR-BOOK-NUM
           END-IF.  
           MOVE RSP06-PAX-SEQ-NUM (IRP04-XFER-SUB)  
                                        TO PTFR-PAX-SEQ-NUM.

           PERFORM DPP17-FETCH-PTFR.

      * IF PTFR RECORD EXISTS AND PURCHASE SOURCE IS 'S' IT SIGNIFIES   
      * TRANSFER IS BOOKED FROM PRE-SALES.  
           IF  DPP03-PTFR-WAS-FETCHED   
               PERFORM VARYING IRP04-PTFR-SUB FROM 1 BY 1   
                         UNTIL IRP04-PTFR-SUB > PTFR-TRANSFER-CNT   
                  IF  PTFR-CANCEL-DATE (IRP04-PTFR-SUB) NOT > ZEROES
                      MOVE 'Y'          TO IRP04-PS-TRANSFER-IND
                      IF  PTFR-PRE-POST-CODE (IRP04-PTFR-SUB) = 'I' 
                          COMPUTE IRP04-PRE-XFR-AMT =   
                                  IRP04-PRE-XFR-AMT +   
                                  PTFR-AMT (IRP04-PTFR-SUB) 
                      ELSE  
                          COMPUTE IRP04-PST-XFR-AMT =   
                                  IRP04-PST-XFR-AMT +   
                                  PTFR-AMT (IRP04-PTFR-SUB) 
                      END-IF
                  END-IF
               END-PERFORM  
           ELSE 
               MOVE 'N'                 TO IRP04-PS-TRANSFER-IND
           END-IF.  

       IRP04-PROCESS-PS-TRANSFERS-EXT.  
           EXIT.

       IRP04-GET-PASSOFF-DATA-EXIT. 
           EXIT.

 END PROC   


IRP04INT PROC   

           MOVE LOW-VALUES              TO IRP04-OUTPUT-DATA.   
           PERFORM VARYING IRP04-PAX-SUB FROM 1 BY 1
                     UNTIL IRP04-PAX-SUB > 5
              MOVE SPACES   TO IRP04-O-PAX-FIRST-NAME   (IRP04-PAX-SUB) 
                               IRP04-O-PAX-LAST-NAME    (IRP04-PAX-SUB) 
                               IRP04-O-PAX-CVP-IND      (IRP04-PAX-SUB) 
                               IRP04-O-PAX-VIP-IND      (IRP04-PAX-SUB) 
                               IRP04-O-PAX-VIFP-NUM     (IRP04-PAX-SUB) 
                               IRP04-O-PAX-VIFP-LEVEL   (IRP04-PAX-SUB) 
                               IRP04-O-PAX-DINING-PREF  (IRP04-PAX-SUB) 
                               IRP04-O-PAX-DINING-CNF   (IRP04-PAX-SUB) 
                               IRP04-O-PAX-PPGRATS-IND  (IRP04-PAX-SUB) 
                               IRP04-O-PAX-OFFER-CODE   (IRP04-PAX-SUB) 
                               IRP04-O-PAX-AIR-TYPE     (IRP04-PAX-SUB) 
                               IRP04-O-PAX-DEPARTING    (IRP04-PAX-SUB) 
                               IRP04-O-PAX-COMMENT-CODE(IRP04-PAX-SUB 1)
                               IRP04-O-PAX-COMMENT-CODE(IRP04-PAX-SUB 2)
                               IRP04-O-PAX-COMMENT-CODE(IRP04-PAX-SUB 3)
                               IRP04-O-PAX-COMMENT-CODE(IRP04-PAX-SUB 4)
                               IRP04-O-PAX-COMMENT-CODE(IRP04-PAX-SUB 5)
                   PERFORM VARYING IRP04-QUAL-SUB FROM 1 BY 1   
                             UNTIL IRP04-QUAL-SUB > 20  
                      MOVE SPACES       TO  
                                          IRP04-O-PAX-QUALIFICATION-CODE
                                          (IRP04-PAX-SUB IRP04-QUAL-SUB)
                   END-PERFORM  
                   PERFORM VARYING IRP04-PPHOR-SUB FROM 1 BY 1  
                             UNTIL IRP04-PPHOR-SUB > 2  
                      MOVE SPACES      TO IRP04-O-PPHOR-EXIST-IND   
                                         (IRP04-PAX-SUB IRP04-PPHOR-SUB)
                                          IRP04-O-PPHOR-PHONE-NUM   
                                         (IRP04-PAX-SUB IRP04-PPHOR-SUB)
                                          IRP04-O-PPHOR-SEND-TXT-PRF-IND
                                         (IRP04-PAX-SUB IRP04-PPHOR-SUB)
                      MOVE ZEROES      TO IRP04-O-PPHOR-SAIL-TXT-OPT-IND
                                         (IRP04-PAX-SUB IRP04-PPHOR-SUB)
                                          IRP04-O-PPHOR-PAYM-TXT-OPT-IND
                                         (IRP04-PAX-SUB IRP04-PPHOR-SUB)
                   END-PERFORM  
                   PERFORM VARYING IRP04-FLT-SUB FROM 1 BY 1
                             UNTIL IRP04-FLT-SUB > 6
                      MOVE ZEROES       TO IRP04-O-FLT-PAX-ID   
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-DEPART-TIME  
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-ARRIVAL-TIME 
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-TICKET-TIME  
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)

                      MOVE SPACES       TO IRP04-O-FLT-DEPART-CITY  
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-ARRIVE-CITY  
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-AIRLINE-NAME 
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-NUMBER   
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-CLASS-OF-SERVICE 
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-ITN-STATUS   
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-TKT-TYPE-FLAG
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-ALT-RECORD-LOC   
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-CABIN-CLASS  
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-OPER-AIRLINE-CODE
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-OPER-AIRLINE-NAME
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-DEP-DAY  
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-AIRLINE-REC-LOC  
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-SEAT-ASSIGNED
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                                           IRP04-O-FLT-TICKET-NUM   
                                           (IRP04-PAX-SUB IRP04-FLT-SUB)
                   END-PERFORM  
           END-PERFORM. 
           PERFORM VARYING IRP04-ITN-SUB FROM 1 BY 1
                   UNTIL IRP04-ITN-SUB > 55 
              MOVE SPACES TO IRP04-O-ITR-DAY-OF-WEEK   (IRP04-ITN-SUB)  
                             IRP04-O-ITR-PORT-OF-CALL  (IRP04-ITN-SUB)  
                             IRP04-O-ITR-ARRIVE-TIME   (IRP04-ITN-SUB)  
                             IRP04-O-ITR-DEPART-TIME   (IRP04-ITN-SUB)  
                             IRP04-O-ITR-TENDER-CODE   (IRP04-ITN-SUB)  
                             IRP04-O-ITR-FOOTNOTE-CODE (IRP04-ITN-SUB)  
           END-PERFORM. 

 END PROC
