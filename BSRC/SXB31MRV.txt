       1120-PROCESS-MAIN.   

           IF WS-ERROR-CODE > SPACES

           END-IF.  

       1140-VALIDATE-INPUT. 

           MOVE ZEROES                 TO WS-COUNT. 
           INSPECT WS-FROM-START-DATE-RANGE TALLYING WS-COUNT FOR   
              ALL '/'   
           IF WS-COUNT = 2  
              CONTINUE  
           ELSE 
              MOVE 'D02'               TO WS-ERROR-CODE 
              GO TO 1120-PROCESS-MAIN-EXIT  
           END-IF.  

           UNSTRING WS-FROM-START-DATE-RANGE
               DELIMITED BY '/' INTO
               WS-MON   
               WS-DD
               WS-YYYY. 

           INSPECT WS-MON REPLACING LEADING SPACES BY '0'.  
           INSPECT WS-DD REPLACING LEADING SPACES BY '0'.   

           MOVE WS-DATE                TO DPP02-YYYYMMDD-DATE.  

           PERFORM DPP10-VALIDATE-YYYYMMDD-DATE.

           IF DPP02-IS-DATE-VALID = 'Y' 
              IF  DPP02-YYYYMMDD-DATE >= WS-SYSTEM-DATE 
                  CONTINUE  
              ELSE  
                  MOVE 'D03'           TO WS-ERROR-CODE 
                  GO TO 1120-PROCESS-MAIN-EXIT  
              END-IF
           ELSE 
              MOVE 'D04'               TO WS-ERROR-CODE 
              GO TO 1120-PROCESS-MAIN-EXIT  
           END-IF.  

           MOVE DPP02-YYYYMMDD-DATE    TO WS-TEMP-START-DATE.   

           MOVE ZEROES                 TO WS-COUNT. 

           IF WS-FROM-END-DATE-RANGE > SPACES   
              CONTINUE  
           ELSE 
              MOVE WS-FROM-START-DATE-RANGE 
                                       TO WS-FROM-END-DATE-RANGE
           END-IF.  

           INSPECT WS-FROM-END-DATE-RANGE TALLYING WS-COUNT FOR 
              ALL '/'   
           IF WS-COUNT = 2  
              CONTINUE  
           ELSE 
              MOVE 'D05'           TO WS-ERROR-CODE 
              GO TO 1120-PROCESS-MAIN-EXIT  
           END-IF.  

           UNSTRING WS-FROM-END-DATE-RANGE  
               DELIMITED BY '/' INTO
               WS-MON   
               WS-DD
               WS-YYYY. 

           INSPECT WS-MON REPLACING LEADING SPACES BY '0'.  
           INSPECT WS-DD REPLACING LEADING SPACES BY '0'.   

           MOVE WS-DATE                TO DPP02-YYYYMMDD-DATE.  

           PERFORM DPP10-VALIDATE-YYYYMMDD-DATE.

           IF DPP02-IS-DATE-VALID = 'Y' 
              IF  DPP02-YYYYMMDD-DATE >= WS-SYSTEM-DATE 
                  CONTINUE  
              ELSE  
                  MOVE 'D06'           TO WS-ERROR-CODE 
                  GO TO 1120-PROCESS-MAIN-EXIT  
              END-IF
           ELSE 
              MOVE 'D07'               TO WS-ERROR-CODE 
              GO TO 1120-PROCESS-MAIN-EXIT  
           END-IF.  

           MOVE DPP02-YYYYMMDD-DATE    TO WS-TEMP-END-DATE. 

           IF WS-TEMP-START-DATE <= WS-TEMP-END-DATE
              CONTINUE  
           ELSE 
              MOVE 'D08'            TO WS-ERROR-CODE
              GO TO 1120-PROCESS-MAIN-EXIT  
           END-IF   

           IF WS-FROM-DAY-OF-WEEK > SPACES  
              MOVE FUNCTION UPPER-CASE (WS-FROM-DAY-OF-WEEK) TO WS-DAY  
              IF WS-VALID-DAY   
                 CONTINUE   
              ELSE  
                 MOVE 'D09'            TO WS-ERROR-CODE 
                 GO TO 1120-PROCESS-MAIN-EXIT   
              END-IF
              MOVE WS-DAY              TO WS-FROM-DAY-OF-WEEK   
           ELSE 
              MOVE 'ALL'               TO WS-FROM-DAY-OF-WEEK   
           END-IF.  

           IF WS-FROM-IGNORE-GUESTS-SCHEDULE > SPACES   
              IF WS-FROM-IGNORE-GUESTS-SCHEDULE = 'Y' OR 'N'
                 CONTINUE   
              ELSE  
                 MOVE 'D10'           TO WS-ERROR-CODE  
              END-IF
           END-IF.  

           IF WS-FROM-IGNORE-PRE-SALE-STATUS > SPACES   
              IF WS-FROM-IGNORE-PRE-SALE-STATUS = 'Y' OR 'N'
                 CONTINUE   
              ELSE  
                 MOVE 'D11'           TO WS-ERROR-CODE  
              END-IF
           END-IF.  

           IF WS-FROM-PORT1 > SPACES
              CONTINUE  
           ELSE 
              MOVE 'ALL'               TO WS-FROM-PORT1 
           END-IF.  

           IF WS-FROM-PRE-SALE-ITEM > SPACES AND
              WS-FROM-PRE-SALE-ITEM NOT = 'ALL' 
              CONTINUE  
           ELSE 
              IF WS-FROM-PORT1 = 'ALL'  
                 MOVE 'D12'            TO WS-ERROR-CODE 
                 GO TO 1120-PROCESS-MAIN-EXIT   
              END-IF
           END-IF.  

           IF WS-FROM-PRE-SALE-ITEM = 'ALL' 
              IF (WS-FROM-EFFECTIVE-SHIP1 NOT = 'ALL' AND SPACES) OR
                 (WS-FROM-EFFECTIVE-SHIP2 NOT = 'ALL' AND SPACES) OR
                 (WS-FROM-EFFECTIVE-SHIP3 NOT = 'ALL' AND SPACES) OR
                 (WS-FROM-EFFECTIVE-SHIP4 NOT = 'ALL' AND SPACES) OR
                 (WS-FROM-EFFECTIVE-SHIP5 NOT = 'ALL' AND SPACES) OR
                 (WS-FROM-EFFECTIVE-SHIP6 NOT = 'ALL' AND SPACES) OR
                 (WS-FROM-EFFECTIVE-SHIP7 NOT = 'ALL' AND SPACES) OR
                 (WS-FROM-EFFECTIVE-SHIP8 NOT = 'ALL' AND SPACES)   
                 CONTINUE   
              ELSE  
                 MOVE 'D40'            TO WS-ERROR-CODE 
                 GO TO 1120-PROCESS-MAIN-EXIT   
              END-IF
           END-IF.  

           IF WS-FROM-PRE-SALE-ITEM > SPACES AND
              WS-FROM-PRE-SALE-ITEM NOT = 'ALL' 
              MOVE WS-FROM-PRE-SALE-ITEM TO EMR-EXC-ID  
              FETCH EXC-MASTER-REC RECORD   
                  ON ERROR GO TO 1180-EMR-NOT-FOUND 
           END-IF.  

           IF (WS-FROM-PORT1 NOT = 'ALL') AND   
              (WS-FROM-PRE-SALE-ITEM > SPACES AND   
               WS-FROM-PRE-SALE-ITEM NOT = 'ALL') AND   
              EMR-PORT-CODE NOT = WS-FROM-PORT1 
              MOVE 'D13'               TO WS-ERROR-CODE 
              GO TO 1120-PROCESS-MAIN-EXIT  
           END-IF.  

           IF WS-FROM-OPERATOR > ZEROES AND 
              WS-FROM-PRE-SALE-ITEM > SPACES AND
              WS-FROM-PRE-SALE-ITEM NOT = 'ALL' 
              IF EMR-VENDOR-ID = WS-FROM-OPERATOR   
                 CONTINUE   
              ELSE  
                 MOVE 'D14'            TO WS-ERROR-CODE 
                 GO TO 1120-PROCESS-MAIN-EXIT   
              END-IF
           END-IF.  

           IF WS-FROM-EFFECTIVE-TIME1 > SPACES  
              MOVE FUNCTION UPPER-CASE (WS-FROM-EFFECTIVE-TIME1)
                                       TO WS-FROM-EFFECTIVE-TIME1   
           END-IF.  
           IF WS-FROM-EFFECTIVE-TIME2 > SPACES  
              MOVE FUNCTION UPPER-CASE (WS-FROM-EFFECTIVE-TIME2)
                                       TO WS-FROM-EFFECTIVE-TIME2   
           END-IF.  
           IF WS-FROM-EFFECTIVE-TIME3 > SPACES  
              MOVE FUNCTION UPPER-CASE (WS-FROM-EFFECTIVE-TIME3)
                                       TO WS-FROM-EFFECTIVE-TIME3   
           END-IF.  
           IF WS-FROM-EFFECTIVE-TIME4 > SPACES  
              MOVE FUNCTION UPPER-CASE (WS-FROM-EFFECTIVE-TIME4)
                                       TO WS-FROM-EFFECTIVE-TIME4   
           END-IF.  

           IF WS-FROM-EFFECTIVE-TIME1 > SPACES AND  
              WS-FROM-EFFECTIVE-TIME1 NOT = 'ALL'   
              UNSTRING WS-FROM-EFFECTIVE-TIME1  
                  DELIMITED BY ':' OR SPACES
                  INTO WS-HH
                       WS-MM
                       WS-AMPM  

              IF WS-AMPM NOT = 'AM' AND 'PM'
                 MOVE 'D15'            TO WS-ERROR-CODE 
                 GO TO 1120-PROCESS-MAIN-EXIT   
              END-IF

              IF WS-AMPM = 'PM' AND WS-HH < 12  
                 COMPUTE WS-HH = WS-HH + 12 
              END-IF

              IF WS-AMPM = 'AM' AND WS-HH = 12  
                 MOVE ZEROES           TO WS-HH 
              END-IF
              INSPECT WS-HH REPLACING ALL SPACES BY '0' 
              MOVE WS-HH               TO WS-TIME1(1:2) 
              MOVE WS-MM               TO WS-TIME1(3:2) 

              IF WS-EFF-TIME1 < 2400
                 CONTINUE   
              ELSE  
                 MOVE 'D15'            TO WS-ERROR-CODE 
                 GO TO 1120-PROCESS-MAIN-EXIT   
              END-IF
           END-IF.  

           IF WS-FROM-EFFECTIVE-TIME2 > SPACES AND  
              WS-FROM-EFFECTIVE-TIME2 NOT = 'ALL'   
              UNSTRING WS-FROM-EFFECTIVE-TIME2  
                  DELIMITED BY ':' OR SPACES
                  INTO WS-HH
                       WS-MM
                       WS-AMPM
