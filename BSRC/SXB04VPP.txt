           PERFORM DPP12-BEGIN-THREAD THRU DPP12-BEGIN-THREAD-EXIT. 

           IF  DPP01-SQL-ERROR-IND = 'Y'
               GO TO DPP12-THREAD-ERROR-EXIT
           END-IF.  

       0100-INIT-DATABASE.  
           IMPART.  

           ACCEPT WS-PARAM-AREA         FROM CARD-READER.   
           MOVE WS-PARAM-AREA           TO EXCINV-CALCAREA(7:2).

           OPEN EXCINV-CALCAREA.

           OPEN INPUT  SKIP-EXC-FILE.   

           MOVE FUNCTION CURRENT-DATE TO WS-DATE-TIME.  

       0200-FETCH-FIRST-ESDR-REC.   
           FETCH FIRST  EXC-SAILING-DEPARTURE-REC   
                 WITHIN EXCINV-CALCAREA AREA
                 ON ERROR GO TO 9100-DATABASE-ERROR.

           GO TO 0400-PROCESS-ESDR. 

       0300-FETCH-NEXT-ESDR-REC.
           FETCH NEXT EXC-SAILING-DEPARTURE-REC 
                 WITHIN EXCINV-CALCAREA AREA
                 AT END   GO TO 9000-END-OF-JOB 
                 ON ERROR GO TO 9100-DATABASE-ERROR.

       0400-PROCESS-ESDR.   

           IF  ESDR-EXC-TYPE = 'EX' 
               CONTINUE 
           ELSE 
               GO TO 0300-FETCH-NEXT-ESDR-REC   
           END-IF.  

215347*    IF  ESDR-STATUS = 'N'
215347*        GO TO 0300-FETCH-NEXT-ESDR-REC   
215347*    END-IF.  

           IF ESDR-SAIL-DATE < WS-DATE  
              GO TO 0300-FETCH-NEXT-ESDR-REC
           END-IF.  

           MOVE ESDR-EXC-ID             TO SKIP-EXC-ID. 
           READ SKIP-EXC-FILE   
                INVALID KEY GO TO 0500-WRITE-OUTPUT.

           GO TO 0300-FETCH-NEXT-ESDR-REC.  

       0500-WRITE-OUTPUT.   

           IF  WS-PARAM-AREA NOT = '01' 
              MOVE 'Y'                  TO SXP99-DONT-WRITE-HEADER  
           END-IF.  

           PERFORM SXP99-WRITE-SX-PRC-FILE THRU 
                   SXP99-WRITE-SX-PRC-FILE-EXIT.

           ADD 1                        TO WS-OUT-VPP-COUNT.

           PERFORM SXP99-WRITE-SX-INV-FILE THRU 
                   SXP99-WRITE-SX-INV-FILE-EXIT.

           ADD 1                        TO WS-OUT-INV-COUNT.

           PERFORM SXP99-WRITE-SX-VPS-FILE THRU 
                   SXP99-WRITE-SX-VPS-FILE-EXIT.

           ADD 1                        TO WS-OUT-VPS-COUNT.

           GO TO 0300-FETCH-NEXT-ESDR-REC.  

       9000-END-OF-JOB. 
           DISPLAY 'TOTAL REC PROCESSED FOR VPP = ' WS-OUT-VPP-COUNT
                                                    UPON PRINTER.   
           DISPLAY 'TOTAL REC PROCESSED FOR INV = ' WS-OUT-INV-COUNT
                                                    UPON PRINTER.   
           DISPLAY 'TOTAL REC PROCESSED FOR VPS = ' WS-OUT-VPS-COUNT
                                                    UPON PRINTER.   

           CLOSE SHOREX-PRICING-FILE
                 SKIP-EXC-FILE  
                 SHOREX-INV-FILE
                 SHOREX-VPS-FILE.   
           GO TO DPP12-END-OF-JOB.
