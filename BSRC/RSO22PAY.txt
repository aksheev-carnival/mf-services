           COPY DPP42INT.   
           MOVE WS-PROG-ID              TO DPP42-HOLD-MSG-SOURCE.   
           MOVE SPACES     TO TDP01-I-DIN-TIME-CODE 
                              TDP01-I-SCAT-CODE 
                              TDP01-I-ACTIVE-IND.   
           MOVE ZEROES     TO TDP01-I-DUR-DAYS  
                              TDP01-I-SAIL-ID.  
           EXEC SQL 
                USE DEFAULT SCHEMA PSTRDMS  
           END-EXEC.

           EXEC SQL WHENEVER SQLERROR   
                CONTINUE
           END-EXEC.

                CONTINUE
           END-EXEC.

           MOVE LOW-VALUES              TO RSP22-O-N-PARAMS

       0100-BASIC-VALIDATION.   

           IF  RSP22-I-SHIP-CODE <= SPACES  
               MOVE 01                  TO DPP42-HOLD-MSG-INDEX 
               GO TO 9200-EDIT-ERROR-MSG
           END-IF.  

           IF  RSP22-I-SAIL-ID <= ZEROES
               MOVE 02                  TO DPP42-HOLD-MSG-INDEX 
               GO TO 9200-EDIT-ERROR-MSG
           END-IF.  

           IF  RSP22-I-REQ-CAB-CODE NOT = 'B' AND 'R'   
               MOVE 03                  TO DPP42-HOLD-MSG-INDEX 
               GO TO 9200-EDIT-ERROR-MSG
           END-IF.  

           IF  RSP22-I-REQ-CAB-CODE = 'B' AND   
               RSP22-I-BERTH-BY-DATE-IND = 'N' AND  
               RSP22-I-OFFER-ID <= ZEROES   
               MOVE 04                  TO DPP42-HOLD-MSG-INDEX 
               GO TO 9200-EDIT-ERROR-MSG
           END-IF.  

           IF  RSP22-I-REQ-CAB-CODE = 'R' AND   
               RSP22-I-POLICY-CODE NOT > SPACES 
               MOVE 05                  TO DPP42-HOLD-MSG-INDEX 
               GO TO 9200-EDIT-ERROR-MSG
           END-IF.  

           IF  RSP22-I-REQ-CAB-CODE = 'B' AND   
               RSP22-I-POLICY-PROMO-CODE > SPACES AND   
               RSP22-I-BERTH-BY-DATE-IND = 'N' AND  
               RSP22-I-BERTH-BY-DATE NOT > SPACES   
               MOVE 06                  TO DPP42-HOLD-MSG-INDEX 
               GO TO 9200-EDIT-ERROR-MSG
           END-IF.  

           IF  RSP22-I-OFFER-ID > ZEROES
               IF  RSP22-I-OCCUPANCY < 1 OR 
                   RSP22-I-OCCUPANCY > 5
                   MOVE 07              TO DPP42-HOLD-MSG-INDEX 
                   GO TO 9200-EDIT-ERROR-MSG
               END-IF   
           END-IF.  

           IF RSP22-I-EXTEND-CODE = 'Y' OR 'M' OR 'N'   
              CONTINUE  
           ELSE 
              MOVE 29                  TO DPP42-HOLD-MSG-INDEX  
              GO TO 9200-EDIT-ERROR-MSG

               END-IF   
               IF  WS-EXT-DATETIME > WS-ORG-DATETIME
                   MOVE WS-EXT-DATETIME 
                                        TO WS-ORG-DATETIME  
               END-IF   
           END-IF.
